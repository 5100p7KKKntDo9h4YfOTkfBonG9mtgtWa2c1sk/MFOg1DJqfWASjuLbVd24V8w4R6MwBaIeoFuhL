const drawBird = () => { const { bird } = gameRef.current; const birdData = birds[selectedBird]; ctx.save(); ctx.translate(bird.x, bird.y); ctx.rotate(bird.rotation * Math.PI / 180); // Bird body with gradient const gradient = ctx.createLinearGradient(-20, -20, 20, 20); setScore(0); } else if (gameState === 'playing') { gameRef.current.bird.velocity = flapPower; playSound(400, 0.1); createParticles(gameRef.current.bird.x, gameRef.current.bird.y, birds[selectedBird].gradient[0]); } else if (gameState === 'gameOver') { setGameState('menu'); } }; canvas.addEventListener('click', handleClick); gameLoop(); return () => { canvas.removeEventListener('click', handleClick); cancelAnimationFrame(animationFrame); }; }, [gameState, sound, selectedBird, selectedWorld]); useEffect(() => { if (gameState === 'gameOver' && score > highScore) { setHighScore(score); } }, [gameState, score, highScore]); const buyItem = (type, index, price) => { if (coins >= price) { setCoins(c => c - price); if (type === 'bird') { setOwnedBirds([...ownedBirds, index]); setSelectedBird(index); } else { setOwnedWorlds([...ownedWorlds, index]); setSelectedWorld(index); } } }; return (
